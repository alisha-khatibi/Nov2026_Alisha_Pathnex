## Jenkins pipeline -NodeJSBuild

learn here how to install dependencies , run test and build NodeJs project



pipeline {
  agent any
  tools {
    nodejs 'NodeJS-16'
}
stages {
  stage('Checkout'){
     steps{
      sh'npm install'

}
}
   stage('Run Tests'){
     steps{
       sh 'npm tests'

}
}
  stage ('Install Dependencies')}{
    steps {
       sh 'npm install'


}    
}

stage('Build'){
steps {

sh 'npm run build'
}
}
}
}
